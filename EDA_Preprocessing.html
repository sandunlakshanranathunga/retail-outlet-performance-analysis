cost_cols = ["Rent_Cost", "Staff_Cost", "Inventory_Cost"]

# Rule 1: exactly one cost column missing, Operating_Cost known -> back-calculate it 
mask1 = df["Operating_Cost"].notna() & (df[cost_cols].isna().sum(axis=1) == 1)
for idx in df[mask1].index:
    missing_col = df.loc[idx, cost_cols][df.loc[idx, cost_cols].isna()].index[0]
    known_sum = df.loc[idx, cost_cols].drop(missing_col).sum()
    df.loc[idx, missing_col] = df.loc[idx, "Operating_Cost"] - known_sum

# Rule 2: Operating_Cost AND Profit_Margin both missing, but all 5 inputs available -> calculate both
mask2 = (df["Operating_Cost"].isna() & df["Profit_Margin"].isna() &
         df[cost_cols + ["Marketing_Spend", "Monthly_Revenue"]].notna().all(axis=1))
df.loc[mask2, "Operating_Cost"] = df.loc[mask2, cost_cols].sum(axis=1)
df.loc[mask2, "Profit_Margin"] = (
    (df.loc[mask2, "Monthly_Revenue"] - df.loc[mask2, "Operating_Cost"] - df.loc[mask2, "Marketing_Spend"])
    / df.loc[mask2, "Monthly_Revenue"]
)

print("Missing values remaining:\n", df.isna().sum())

# Rent_Cost is fairly symmetric, ranging roughly 50–400, with most outlets paying between 150–300. No major skew.
# Staff_Cost is roughly bell-shaped, centered around 300–400, with a slight spread up to 600. Reflects normal variation in staffing size across outlets.
# Inventory_Cost is roughly symmetric and centered around 500–600, with a wider spread than the other cost components — consistent with inventory needs varying more by outlet type (e.g. Wholesale vs Convenience).
# Operating_Cost (the sum of the three cost components) is right-skewed, clustering around 800–1200 with a tail extending to 1800 — driven by a smaller group of higher-cost outlets.
# Marketing_Spend is right-skewed, with most outlets spending 50–150 and a long tail up to 250. Few outlets invest heavily in marketing.
# Monthly_Revenue is right-skewed, with most outlets earning 1500–3500 and a smaller number of high-performing outlets pulling the distribution toward 5000. This skew is why we picked it as our dependent variable — it's the clearest signal of performance variation across outlets.
# Profit_Margin is heavily left-skewed, with most outlets clustered tightly around 0.5–0.75, but a small group of outliers showing strongly negative margins (down to -1.0) — likely outlets whose operating and marketing costs exceeded revenue.
